# Scheduled-trigger runtime (v0.27). MemoryJobStore is enough — we re-
# register every scheduled agent from the agents table on boot, so we
# don't need APScheduler's SQLAlchemyJobStore (which would pull in
# SQLAlchemy as a transitive dep). croniter is already pulled in by
# dialekt-manifest-validator for cron validation; APScheduler reuses it.
apscheduler>=3.10,<4
croniter>=2.0.0
# All published sqlite-vec versions cap at python<3.12. On most platforms
# pip pulls a prebuilt wheel that ignores that bound, but Windows ARM64
# has no wheel, so pip falls back to the metadata-respecting sdist and
# fails. Skip on win-arm64 — sqlite_vec is only present as a
# precautionary hidden import for PyInstaller, no source code uses it yet.
sqlite-vec>=0.1.9; platform_machine != "ARM64"
# Model Context Protocol SDK — consumed by dialekt.mcp.client to talk to
# external MCP servers and by dialekt.mcp.server to expose dialekt as one.
# Pinned <1.27 because 1.27 transitively pulls starlette>=1.0 via
# sse-starlette>=3.x, which is incompatible with the current fastapi
# 0.115.2 + open-interpreter 0.4.3 pair (both cap starlette<0.38).
# sse-starlette<3 is pinned explicitly to defend against transitive drift.
# Revisit the upper bound when fastapi + open-interpreter accept starlette>=1.0.
# Rationale: docs/M2_MCP_DESIGN.md Decision 1 addendum.
mcp>=1.25,<1.27
sse-starlette<3

# Visual content engine (HTML → PNG via headless Chromium). After
# installing Python deps, run `playwright install chromium` once to
# fetch the browser binaries (~150MB). Pillow ships with PIL.Image
# which the engine uses for read-back / dimension probes and which
# agents can call from their Python code blocks for ad-hoc image ops.
playwright>=1.50,<2
Pillow>=10.0

# RSS / Atom feed parser used by the MCP rss tool and the scheduler's
# rss-poll integration. Lightweight, pure-Python, no native deps.
feedparser>=6.0,<7

# S3-compatible storage client for the cdn upload MCP tool. Works with
# AWS S3, Backblaze B2, Cloudflare R2, MinIO — anything that speaks
# the S3 API. Without this Instagram publish can't run end-to-end
# (Graph API requires public HTTPS URLs for media).
boto3>=1.34,<2
